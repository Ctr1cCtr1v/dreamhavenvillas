{{/* Photo grid: one large photo + up to two smaller ones; a "+N" badge shows when there are more. Every photo links to a large version and carries data-lightbox so the gallery viewer can open the whole set. Usage: {{ partial "photo-grid.html" (dict "photos" .photos "name" .name "group" "living-room") }} Optional: sizes (for the large photo) */}} {{- $photos := .photos | default slice }} {{- $name := .name }} {{- $group := .group | default (urlize $name) }} {{- $count := len $photos }} {{- if $count }}
{{- range $i, $src := $photos }} {{- with resources.Get (printf "images/%s" $src) }} {{- $big := . }} {{- if gt .Width 2000 }}{{ $big = .Resize "2000x webp" }}{{ else }}{{ $big = .Resize (printf "%dx webp" .Width) }}{{ end }} {{- $alt := printf "%s — photo %d" $name (add $i 1) }} {{- if lt $i 3 }} {{ partial "img.html" (dict "src" $src "alt" $alt "sizes" (cond (eq $i 0) ($.sizes | default "(min-width: 860px) 40vw, 100vw") "(min-width: 860px) 20vw, 50vw")) }} {{- if and (eq $i 2) (gt $count 3) }}+{{ sub $count 3 }} photos{{ end }} {{- end }} {{- else }} {{- warnf "photo-grid: image not found assets/images/%s" $src }} {{- end }} {{- end }}
{{- end }}
